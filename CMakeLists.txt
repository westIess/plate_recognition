cmake_minimum_required(VERSION 3.16)
project(plate_recognition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(CTest)
option(PLATE_BUILD_RECOGNIZER "Build OpenCV/Tesseract pipeline and generator" ON)
option(PLATE_ENABLE_CASCADE "Use cascade if its OpenCV module is available" ON)

add_library(plate_common INTERFACE)
target_include_directories(plate_common INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(evaluate src/evaluate.cpp)
target_link_libraries(evaluate PRIVATE plate_common) # Deliberately no OpenCV/OCR dependency.
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.cpp)
    target_link_libraries(test_core PRIVATE plate_common)
    add_test(NAME core COMMAND test_core)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME evaluation COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_evaluate.py" "$<TARGET_FILE:evaluate>")
    endif()
endif()

if(PLATE_BUILD_RECOGNIZER)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
    if(OpenCV_VERSION_MAJOR LESS 4)
        message(FATAL_ERROR "OpenCV >= 4 is required")
    endif()
    set(plate_cv_components core imgproc imgcodecs)
    if(OpenCV_VERSION_MAJOR GREATER_EQUAL 5)
        list(APPEND plate_cv_components geometry)
        set(plate_cascade_component xobjdetect)
    else()
        set(plate_cascade_component objdetect)
    endif()
    if(PLATE_ENABLE_CASCADE)
        find_package(OpenCV REQUIRED COMPONENTS ${plate_cv_components}
            OPTIONAL_COMPONENTS ${plate_cascade_component})
    else()
        find_package(OpenCV REQUIRED COMPONENTS ${plate_cv_components})
    endif()
    set(plate_has_cascade 0)
    if(PLATE_ENABLE_CASCADE AND OpenCV_${plate_cascade_component}_FOUND)
        set(plate_has_cascade 1)
    endif()
    message(STATUS "OpenCV ${OpenCV_VERSION}; cascade support: ${plate_has_cascade}")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(TESSERACT REQUIRED IMPORTED_TARGET tesseract)
    pkg_check_modules(LEPTONICA REQUIRED IMPORTED_TARGET lept)
    add_library(plate_vision src/plate_detector.cpp)
    target_link_libraries(plate_vision PUBLIC plate_common ${OpenCV_LIBS})
    target_include_directories(plate_vision SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
    target_compile_definitions(plate_vision PUBLIC PLATE_HAS_CASCADE=${plate_has_cascade})
    add_executable(plate_recognizer src/main.cpp src/ocr_reader.cpp)
    target_link_libraries(plate_recognizer PRIVATE plate_vision PkgConfig::TESSERACT PkgConfig::LEPTONICA)
    add_executable(generate_test_image src/generate_test_image.cpp)
    target_link_libraries(generate_test_image PRIVATE plate_vision)
    if(BUILD_TESTING)
        add_executable(test_geometry tests/test_geometry.cpp)
        target_link_libraries(test_geometry PRIVATE plate_vision)
        add_test(NAME geometry COMMAND test_geometry)
    endif()
endif()
